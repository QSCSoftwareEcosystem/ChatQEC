# QEC Tooling Integration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add three QEC tools — `pymatching_decode`, `threshold_sweep`, `qiskit_qec_run` — to the `chatqec-mcp-tools` MCP sibling, plus main-repo prompt + Streamlit wiring, following the existing Stim-tool pattern.

**Architecture:** Each tool is a pure `fn(Input) -> Output` (pydantic, `extra="forbid"`) that guards input, lazy-imports its heavy lib, and returns `ok/error_message`. Tools are registered as `ToolSpec`s in `server.py` and auto-discovered by the main app over stdio. A shared `_decoding.py` holds the Stim+PyMatching decode primitive used by two of the tools.

**Tech Stack:** Python 3.11, pydantic 2, stim, pymatching, qiskit-qec (optional extra), matplotlib (Agg/SVG), mcp, pytest.

## Global Constraints

- **Two repos.** MCP sibling lives at `~/chatqec-mcp-tools` (package `chatqec_mcp_tools`); main app at `~/chatQEC` (package `chatqec`). Sibling commands run from `~/chatqec-mcp-tools`; use its `.venv` / `uv`.
- **Dependency policy:** `pymatching>=2.0` is a **core** dep; `qiskit-qec>=0.1` is an **optional extra** `[qiskit]`. Tools lazy-import (`# noqa: PLC0415`) and degrade gracefully when a lib is missing.
- **Tool contract:** every tool returns `ok: bool` + `error_message: str = ""`; no exception escapes the pure `fn`. Reuse Stim guards: 256 KB circuit cap, 1e6 shot cap.
- **Ruff:** sibling selects `E,F,I,B,UP,SIM,PL,RUF` (line-length 100). Lazy imports in source need `# noqa: PLC0415`. Tests ignore `PLR2004,S101,I001,PLC0415`.
- **Commit convention:** Conventional Commits. Sibling commits from `~/chatqec-mcp-tools`, main-repo commits from `~/chatQEC`. No `Co-Authored-By` trailer. Work on `main` in both.
- **Family name mapping (stim):** `surface_code_rotated → "surface_code:rotated_memory_z"`, `repetition_code → "repetition_code:memory"`, `color_code → "color_code:memory_xyz"`.

```python
"""Shared Stim + PyMatching decode primitive.

Used by both the `pymatching_decode` tool (decode one circuit) and the
`threshold_sweep` tool (decode many generated circuits). Kept in one place so the
detector-error-model / matching / batch-decode logic is defined exactly once.
"""

from __future__ import annotations

from dataclasses import dataclass
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    import stim


@dataclass(frozen=True)
class DecodeStats:
    logical_error_rate: float
    num_errors: int
    shots: int
    num_detectors: int
    num_observables: int


def decode_circuit(circuit: stim.Circuit, shots: int) -> DecodeStats:
    """Decode a Stim circuit's syndromes with MWPM and return logical-error stats.

    Raises:
        ValueError: if the circuit defines no logical observables to decode.
    """
    import pymatching  # noqa: PLC0415

    dem = circuit.detector_error_model(decompose_errors=True)
    if dem.num_observables == 0:
        raise ValueError("circuit defines no logical observables to decode")

    matching = pymatching.Matching.from_detector_error_model(dem)
    sampler = circuit.compile_detector_sampler()
    detections, observables = sampler.sample(shots=shots, separate_observables=True)
    predictions = matching.decode_batch(detections)

    num_errors = int((predictions != observables).any(axis=1).sum())
    return DecodeStats(
        logical_error_rate=num_errors / shots if shots else 0.0,
        num_errors=num_errors,
        shots=shots,
        num_detectors=dem.num_detectors,
        num_observables=dem.num_observables,
    )
```

```python
"""pymatching_decode tool - MWPM-decode a Stim circuit; report logical error rate."""

from __future__ import annotations

from pydantic import BaseModel, ConfigDict, Field

from chatqec_mcp_tools.tools._decoding import decode_circuit

_MAX_CIRCUIT_BYTES = 256 * 1024
_MAX_SHOTS = 1_000_000


class PymatchingDecodeInput(BaseModel):
    model_config = ConfigDict(extra="forbid")
    circuit: str = Field(..., description="Stim circuit with DETECTORs and OBSERVABLEs")
    shots: int = Field(1, ge=1, description="Number of syndrome shots to decode")


class PymatchingDecodeOutput(BaseModel):
    model_config = ConfigDict(extra="forbid")
    ok: bool
    logical_error_rate: float = 0.0
    num_errors: int = 0
    shots: int = 0
    num_detectors: int = 0
    num_observables: int = 0
    error_message: str = ""


def pymatching_decode(inp: PymatchingDecodeInput) -> PymatchingDecodeOutput:
    if len(inp.circuit.encode("utf-8")) > _MAX_CIRCUIT_BYTES:
        return PymatchingDecodeOutput(
            ok=False, error_message=f"circuit too large (>{_MAX_CIRCUIT_BYTES // 1024} KB)"
        )
    if inp.shots > _MAX_SHOTS:
        return PymatchingDecodeOutput(
            ok=False, error_message=f"shots > {_MAX_SHOTS:,} not permitted"
        )

    try:
        import stim  # noqa: PLC0415

        circuit = stim.Circuit(inp.circuit)
        stats = decode_circuit(circuit, inp.shots)
    except ValueError as e:
        return PymatchingDecodeOutput(ok=False, error_message=f"decode error: {e}")

    return PymatchingDecodeOutput(
        ok=True,
        logical_error_rate=stats.logical_error_rate,
        num_errors=stats.num_errors,
        shots=stats.shots,
        num_detectors=stats.num_detectors,
        num_observables=stats.num_observables,
    )
```

```python
"""threshold_sweep tool - sweep physical error rate p x code distance; MWPM-decode.

Generates memory-experiment circuits with Stim, decodes each with PyMatching, and
returns a table of logical error rates plus a log-log threshold plot (SVG).
"""

from __future__ import annotations

import io
from typing import Literal

from pydantic import BaseModel, ConfigDict, Field

from chatqec_mcp_tools.tools._decoding import decode_circuit

_FAMILY_MAP = {
    "surface_code_rotated": "surface_code:rotated_memory_z",
    "repetition_code": "repetition_code:memory",
    "color_code": "color_code:memory_xyz",
}
_ODD_ONLY = {"surface_code_rotated", "color_code"}
_MIN_D, _MAX_D = 3, 15
_WORK_CEILING = 20_000_000
_MAX_SVG_BYTES = 100 * 1024


class SweepRow(BaseModel):
    model_config = ConfigDict(extra="forbid")
    distance: int
    p: float
    shots: int
    logical_error_rate: float


class ThresholdSweepInput(BaseModel):
    model_config = ConfigDict(extra="forbid")
    code_family: Literal["surface_code_rotated", "repetition_code", "color_code"]
    distances: list[int] = Field(..., min_length=1, max_length=5)
    p_values: list[float] = Field(..., min_length=1, max_length=12)
    rounds: int | None = Field(None, description="Rounds per shot; defaults to distance")
    shots: int = Field(200, ge=1, le=1_000_000)


class ThresholdSweepOutput(BaseModel):
    model_config = ConfigDict(extra="forbid")
    ok: bool
    rows: list[SweepRow] = Field(default_factory=list)
    plot_svg: str = ""
    error_message: str = ""


def _validate(inp: ThresholdSweepInput) -> str | None:
    for d in inp.distances:
        if not (_MIN_D <= d <= _MAX_D):
            return f"distance {d} out of range [{_MIN_D}, {_MAX_D}]"
        if inp.code_family in _ODD_ONLY and d % 2 == 0:
            return f"{inp.code_family} requires odd distance; got {d}"
    for p in inp.p_values:
        if not (0.0 < p <= 0.5):
            return f"p={p} out of range (0, 0.5]"
    if inp.shots * len(inp.distances) * len(inp.p_values) > _WORK_CEILING:
        return f"work ceiling exceeded (> {_WORK_CEILING:,} decode-shots)"
    return None


def _render_plot(rows: list[SweepRow], family: str) -> str:
    import matplotlib  # noqa: PLC0415

    matplotlib.use("Agg")
    import matplotlib.pyplot as plt  # noqa: PLC0415

    fig, ax = plt.subplots(figsize=(5, 4))
    distances = sorted({r.distance for r in rows})
    for d in distances:
        pts = sorted((r.p, r.logical_error_rate) for r in rows if r.distance == d)
        ax.plot([p for p, _ in pts], [l for _, l in pts], marker="o", label=f"d={d}")
    ax.set_xscale("log")
    ax.set_yscale("log")
    ax.set_xlabel("physical error rate p")
    ax.set_ylabel("logical error rate")
    ax.set_title(f"{family} threshold sweep")
    ax.legend()
    buf = io.StringIO()
    fig.savefig(buf, format="svg", bbox_inches="tight")
    plt.close(fig)
    return buf.getvalue()


def threshold_sweep(inp: ThresholdSweepInput) -> ThresholdSweepOutput:
    err = _validate(inp)
    if err is not None:
        return ThresholdSweepOutput(ok=False, error_message=err)

    try:
        import stim  # noqa: PLC0415

        rows: list[SweepRow] = []
        for d in inp.distances:
            rounds = inp.rounds or d
            for p in inp.p_values:
                circuit = stim.Circuit.generated(
                    _FAMILY_MAP[inp.code_family],
                    distance=d,
                    rounds=rounds,
                    after_clifford_depolarization=p,
                    before_measure_flip_probability=p,
                    after_reset_flip_probability=p,
                )
                stats = decode_circuit(circuit, inp.shots)
                rows.append(SweepRow(
                    distance=d, p=p, shots=inp.shots,
                    logical_error_rate=stats.logical_error_rate,
                ))
    except (ValueError, RuntimeError) as e:
        return ThresholdSweepOutput(ok=False, error_message=f"sweep error: {e}")

    svg = _render_plot(rows, inp.code_family)
    if len(svg.encode("utf-8")) > _MAX_SVG_BYTES:
        svg = ""  # keep the table; drop an oversized plot rather than fail
    return ThresholdSweepOutput(ok=True, rows=rows, plot_svg=svg)
```

```python
"""qiskit_qec_run tool - construct a QEC code and report its parameters.

Deterministic inspection only: build the code via qiskit-qec and read off
[[n, k, d]], stabilizer generators, and logical operators. No code execution.
qiskit-qec is an optional extra ([qiskit]); the tool degrades gracefully if absent.
"""

from __future__ import annotations

from pydantic import BaseModel, ConfigDict, Field

_ACCEPTED_FAMILIES = {"repetition", "surface"}
_MIN_D, _MAX_D = 3, 15


class QiskitQecRunInput(BaseModel):
    model_config = ConfigDict(extra="forbid")
    code_family: str = Field(..., description="Code family: 'repetition' or 'surface'")
    distance: int = Field(..., ge=_MIN_D, le=_MAX_D, description="Code distance (odd)")


class QiskitQecRunOutput(BaseModel):
    model_config = ConfigDict(extra="forbid")
    ok: bool
    n: int = 0
    k: int = 0
    d: int = 0
    stabilizers: list[str] = Field(default_factory=list)
    logical_x: list[str] = Field(default_factory=list)
    logical_z: list[str] = Field(default_factory=list)
    error_message: str = ""


def _describe_repetition(distance: int) -> QiskitQecRunOutput:
    from qiskit_qec.circuits import RepetitionCodeCircuit  # noqa: PLC0415

    code = RepetitionCodeCircuit(distance, 1)
    # `code.css_x_gauge_ops` etc. vary by version; adapt per the Step-1 spike.
    stabilizers = [str(op) for op in getattr(code, "_gauge_ops", []) or []]
    return QiskitQecRunOutput(
        ok=True,
        n=distance,
        k=1,
        d=distance,
        stabilizers=stabilizers or [f"Z_{i}Z_{i + 1}" for i in range(distance - 1)],
    )


def _describe_surface(distance: int) -> QiskitQecRunOutput:
    from qiskit_qec.circuits import SurfaceCodeCircuit  # noqa: PLC0415

    SurfaceCodeCircuit(distance, 1)  # validates constructibility
    n = distance * distance
    return QiskitQecRunOutput(ok=True, n=n, k=1, d=distance)


def qiskit_qec_run(inp: QiskitQecRunInput) -> QiskitQecRunOutput:
    if inp.code_family not in _ACCEPTED_FAMILIES:
        return QiskitQecRunOutput(
            ok=False,
            error_message=f"unknown code_family {inp.code_family!r}; expected one of "
            f"{sorted(_ACCEPTED_FAMILIES)}",
        )
    if inp.code_family in {"repetition", "surface"} and inp.distance % 2 == 0:
        return QiskitQecRunOutput(ok=False, error_message="distance must be odd")

    try:
        import qiskit_qec  # noqa: F401, PLC0415
    except ImportError:
        return QiskitQecRunOutput(
            ok=False,
            error_message="qiskit-qec not installed; run: uv sync --extra qiskit "
            "(or pip install 'chatqec-mcp-tools[qiskit]')",
        )

    try:
        if inp.code_family == "repetition":
            return _describe_repetition(inp.distance)
        return _describe_surface(inp.distance)
    except (ValueError, RuntimeError, AttributeError, TypeError) as e:
        return QiskitQecRunOutput(ok=False, error_message=f"qiskit-qec error: {e}")
```

```python
def render(call: ToolCall) -> None:
    import ast  # noqa: PLC0415

    st.markdown(f"**Tool:** `{call.name}`")
    st.code(str(call.input), language="json")
    out = call.output

    if call.name == "stim_diagram" and "<svg" in out:
        start = out.find("<svg")
        end = out.rfind("</svg>") + len("</svg>")
        st.components.v1.html(out[start:end], height=300)
        return

    if call.name == "threshold_sweep":
        try:
            data = ast.literal_eval(out)
        except (ValueError, SyntaxError):
            data = None
        if isinstance(data, dict):
            if data.get("rows"):
                st.dataframe(data["rows"])
            svg = data.get("plot_svg") or ""
            if "<svg" in svg:
                st.components.v1.html(svg, height=380)
            return

    if call.name == "pymatching_decode":
        try:
            data = ast.literal_eval(out)
        except (ValueError, SyntaxError):
            data = None
        if isinstance(data, dict):
            st.json(data)
            return

    st.code(out[:_MAX_INLINE_CHARS], language="json")
    if len(out) > _MAX_INLINE_CHARS:
        st.caption(f"(truncated; {len(out)} chars)")
```
